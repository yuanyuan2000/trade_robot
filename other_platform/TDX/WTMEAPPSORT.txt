{公式名称：WTMEAPPSORT}
{公式类型：技术指标公式 / 副图；只在日线周期使用}
{PC/APP跨证券看板辅助指标；只有1个输出SORTV}
{NN、HH、LB、DP在编辑器参数表中设置；缺省值依次为13、6、5、5}
{EPS固定为0.00000001}

EPS:=0.00000001;

PC:=REF(C,1);
RET:=IF(PC>0,C/PC-1,0);
FULLTR:=MAX(MAX(H-L,ABS(H-PC)),ABS(L-PC));
POINTTR:=IF(ISLASTBAR,ABS(C-PC),FULLTR);
NTR:=IF(PC>0,POINTTR/PC,0);

DECAY:=POW(2,-1/HH);
ALPHA:=1-DECAY;
DRET:=DMA(RET,ALPHA);
DNTR:=DMA(NTR,ALPHA);
TAIL:=POW(DECAY,NN);
RW0:=(DRET-TAIL*REF(DRET,NN))/(1-TAIL);
AW0:=(DNTR-TAIL*REF(DNTR,NN))/(1-TAIL);

VALID:=BARSCOUNT(C)>MAX(NN,LB) AND PC>0;
WTME0:=100*RW0/(AW0+EPS);
DROP0:=COUNT(RET<=-DP/100,LB)>0;

SORTV:IF(VALID AND DROP0=0,WTME0,-999),COLORRED,LINETHICK2;
