{公式名称：WTMEAPPSORT}
{公式类型：技术指标公式 / 副图；只在日线周期使用}
{PC/APP跨证券看板辅助指标；参数固定为当前实际WTME策略设置}
{WTME 13/6，急跌回看5段/阈值5%，买入需排名前1且评分>-15}
{动态杠杆VOLAT15、压力10日、最大损失40%、上限5，向下取0.1倍}
{本公式也是跨证券看板的数据接口，最后四行的顺序和名称不能改动}

EPS:=0.00000001;
NN0:=13;
HH0:=6;
LB0:=5;
DP0:=5;
VT0:=15;
SD0:=10;
LOSS0:=40;
ML0:=5;

{一、计算单段收益RET和标准化真实波幅NTR}
PC:=REF(C,1);
RET:=IF(PC>0,C/PC-1,0);
FULLTR:=MAX(MAX(H-L,ABS(H-PC)),ABS(L-PC));
POINTTR:=IF(ISLASTBAR,ABS(C-PC),FULLTR);
NTR:=IF(PC>0,POINTTR/PC,0);

{二、用半衰期衰减和有限窗口修正，得到13段加权收益与波幅}
DECAY:=POW(2,-1/HH0);
ALPHA:=1-DECAY;
DRET:=DMA(RET,ALPHA);
DNTR:=DMA(NTR,ALPHA);
TAIL:=POW(DECAY,NN0);
RW0:=(DRET-TAIL*REF(DRET,NN0))/(1-TAIL);
AW0:=(DNTR-TAIL*REF(DNTR,NN0))/(1-TAIL);

{三、WTME=加权收益/加权波幅；并检查最近5段是否跌幅达到5%}
VALID:=BARSCOUNT(C)>MAX(NN0,LB0) AND PC>0;
WTME0:=100*RW0/(AW0+EPS);
DROP0:=COUNT(RET<=-DP0/100,LB0)>0;

{四、用15日年化波动率和10日压力期计算动态杠杆}
{杠杆限制在1至5倍，并向下取整到0.1倍}
VVALID:=BARSCOUNT(C)>VT0 AND PC>0;
LOGRET:=IF(PC>0,LN(C/PC),0);
HV0:=STD(LOGRET,VT0)*SQRT(252)*100;
STRESS0:=HV0*3*SQRT(SD0/252);
RAWLEV:=IF(STRESS0=0,ML0,LOSS0/STRESS0);
BOUNDLEV:=MIN(ML0,MAX(1,RAWLEV));
DLEV0:=INTPART((BOUNDLEV+0.000000000001)*10)/10;

{五、公开给看板的固定接口；CALCSTOCKINDEX按下列行号读取}
{1=SORTV过滤后分数；2=WTMEV原始分数；3=ELIGV资格；4=DLEV动态杠杆}
SORTV:IF(VALID AND DROP0=0,WTME0,-100),COLORRED,LINETHICK2;
WTMEV:IF(VALID,WTME0,-100),NODRAW;
ELIGV:IF(VALID AND DROP0=0,1,0),NODRAW;
DLEV:IF(VVALID,DLEV0,0),NODRAW;
