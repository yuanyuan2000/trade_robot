{公式名称：WTMES}
{公式类型：技术指标公式 / 副图；只在日线周期使用}
{输出顺序固定：1 WTME，2 RW，3 AW，4 RDROP，5 ELIG，6 SORTV}

{NN、HH、LB、DP 在编辑器参数1-4表中设置}
{缺省值依次为13、6、5、5；EPS固定不开放}
EPS:=0.00000001;

PC:=REF(C,1);
RET:=IF(PC>0,C/PC-1,0);
FULLTR:=MAX(MAX(H-L,ABS(H-PC)),ABS(L-PC));
POINTTR:=IF(ISLASTBAR,ABS(C-PC),FULLTR);
NTR:=IF(PC>0,POINTTR/PC,0);

DECAY:=POW(2,-1/HH);
ALPHA:=1-DECAY;
DRET:=DMA(RET,ALPHA);
DNTR:=DMA(NTR,ALPHA);
TAIL:=POW(DECAY,NN);
RW0:=(DRET-TAIL*REF(DRET,NN))/(1-TAIL);
AW0:=(DNTR-TAIL*REF(DNTR,NN))/(1-TAIL);

VALID:=BARSCOUNT(C)>MAX(NN,LB) AND PC>0;
WTME0:=100*RW0/(AW0+EPS);
DROP0:=COUNT(RET<=-DP/100,LB)>0;

WTME:IF(VALID,WTME0,DRAWNULL),COLORRED,LINETHICK2;
RW:IF(VALID,RW0,DRAWNULL),NODRAW;
AW:IF(VALID,AW0,DRAWNULL),NODRAW;
RDROP:IF(VALID,DROP0,DRAWNULL),NODRAW;
ELIG:IF(VALID AND DROP0=0,1,0),NODRAW;
SORTV:IF(VALID AND DROP0=0,WTME0,-999),NODRAW;
