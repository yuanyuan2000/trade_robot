{公式名称：WTMEPOOLDASH}
{公式类型：技术指标公式 / 副图；只在日线周期使用}
{PC固定池看板：不执行条件选股，直接跨品种读取5个标的的WTMEAPPSORT}
{代码：黄金518880、白银LOF161226、南方原油LOF501018、纳指513100、创业板159915}

GOLD0:=CALCSTOCKINDEX('SH518880','WTMEAPPSORT',1);
SILVER0:=CALCSTOCKINDEX('SZ161226','WTMEAPPSORT',1);
OIL0:=CALCSTOCKINDEX('SH501018','WTMEAPPSORT',1);
NASDAQ0:=CALCSTOCKINDEX('SH513100','WTMEAPPSORT',1);
GEM0:=CALCSTOCKINDEX('SZ159915','WTMEAPPSORT',1);

GOLD1:=IF(ISVALID(GOLD0),GOLD0,-100);
SILVER1:=IF(ISVALID(SILVER0),SILVER0,-100);
OIL1:=IF(ISVALID(OIL0),OIL0,-100);
NASDAQ1:=IF(ISVALID(NASDAQ0),NASDAQ0,-100);
GEM1:=IF(ISVALID(GEM0),GEM0,-100);

BEST0:=MAX(MAX(MAX(MAX(GOLD1,SILVER1),OIL1),NASDAQ1),GEM1);
{完全同分时按证券代码升序：159915、161226、501018、513100、518880}
TOPID:=IF(GEM1=BEST0,1,IF(SILVER1=BEST0,2,IF(OIL1=BEST0,3,IF(NASDAQ1=BEST0,4,5))));
HASBEST:=BEST0>-100;

BEST:IF(HASBEST,BEST0,DRAWNULL),COLORRED,LINETHICK2;
GOLD:GOLD1,NODRAW;
SILVER:SILVER1,NODRAW;
OIL:OIL1,NODRAW;
NASDAQ:NASDAQ1,NODRAW;
GEM:GEM1,NODRAW;

DRAWTEXT_FIX(ISLASTBAR,0.02,0.05,0,'急跌过滤后WTME最高：'),COLORWHITE;
DRAWTEXT_FIX(ISLASTBAR AND HASBEST AND TOPID=1,0.30,0.05,0,'创业板ETF 159915'),COLORYELLOW;
DRAWTEXT_FIX(ISLASTBAR AND HASBEST AND TOPID=2,0.30,0.05,0,'白银LOF 161226'),COLORYELLOW;
DRAWTEXT_FIX(ISLASTBAR AND HASBEST AND TOPID=3,0.30,0.05,0,'南方原油LOF 501018'),COLORYELLOW;
DRAWTEXT_FIX(ISLASTBAR AND HASBEST AND TOPID=4,0.30,0.05,0,'纳指ETF 513100'),COLORYELLOW;
DRAWTEXT_FIX(ISLASTBAR AND HASBEST AND TOPID=5,0.30,0.05,0,'黄金ETF 518880'),COLORYELLOW;
DRAWTEXT_FIX(ISLASTBAR AND NOT(HASBEST),0.30,0.05,0,'当前无合格标的'),COLORGREEN;
DRAWNUMBER_FIX(ISLASTBAR AND HASBEST,0.62,0.05,0,BEST0),COLORYELLOW;

DRAWTEXT_FIX(ISLASTBAR,0.02,0.18,0,'黄金'),COLORWHITE;
DRAWNUMBER_FIX(ISLASTBAR,0.12,0.18,0,GOLD1),COLORWHITE;
