{公式类型：移动端技术指标公式 / 副图；只在日线周期使用}

{必须先在指标平台创建并上传WTMEAPPSORT；本公式不使用CALCSTOCKINDEX和ISVALID}
{当前策略：WTME13/6，急跌5段/5%，第一名且评分严格大于-15}
{BTC/USD以IBIT替代，US10Y不引入；美股市场号为74}

{一、移动端用代码$指标.输出名读取过滤后WTME和动态杠杆}

{二、按过滤后的WTME降序生成唯一名次；完全同分按代码字母顺序}
RD:=1+(FXI1>DRAM1)+(GLD1>DRAM1)+(IBIT1>DRAM1)+(QQQ1>DRAM1)+(SLV1>DRAM1)+(SOXX1>DRAM1)+(SPY1>DRAM1)+(XLE1>DRAM1);
RF:=1+(DRAM1>=FXI1)+(GLD1>FXI1)+(IBIT1>FXI1)+(QQQ1>FXI1)+(SLV1>FXI1)+(SOXX1>FXI1)+(SPY1>FXI1)+(XLE1>FXI1);
RG:=1+(DRAM1>=GLD1)+(FXI1>=GLD1)+(IBIT1>GLD1)+(QQQ1>GLD1)+(SLV1>GLD1)+(SOXX1>GLD1)+(SPY1>GLD1)+(XLE1>GLD1);
RI:=1+(DRAM1>=IBIT1)+(FXI1>=IBIT1)+(GLD1>=IBIT1)+(QQQ1>IBIT1)+(SLV1>IBIT1)+(SOXX1>IBIT1)+(SPY1>IBIT1)+(XLE1>IBIT1);
RQ:=1+(DRAM1>=QQQ1)+(FXI1>=QQQ1)+(GLD1>=QQQ1)+(IBIT1>=QQQ1)+(SLV1>QQQ1)+(SOXX1>QQQ1)+(SPY1>QQQ1)+(XLE1>QQQ1);
RL:=1+(DRAM1>=SLV1)+(FXI1>=SLV1)+(GLD1>=SLV1)+(IBIT1>=SLV1)+(QQQ1>=SLV1)+(SOXX1>SLV1)+(SPY1>SLV1)+(XLE1>SLV1);
RO:=1+(DRAM1>=SOXX1)+(FXI1>=SOXX1)+(GLD1>=SOXX1)+(IBIT1>=SOXX1)+(QQQ1>=SOXX1)+(SLV1>=SOXX1)+(SPY1>SOXX1)+(XLE1>SOXX1);
RS:=1+(DRAM1>=SPY1)+(FXI1>=SPY1)+(GLD1>=SPY1)+(IBIT1>=SPY1)+(QQQ1>=SPY1)+(SLV1>=SPY1)+(SOXX1>=SPY1)+(XLE1>SPY1);
RX:=1+(DRAM1>=XLE1)+(FXI1>=XLE1)+(GLD1>=XLE1)+(IBIT1>=XLE1)+(QQQ1>=XLE1)+(SLV1>=XLE1)+(SOXX1>=XLE1)+(SPY1>=XLE1);

{三、第一名且评分大于-15为建议标的；敞口=100%仓位乘动态杠杆}
DSEL:=RD=1 AND DRAM1>-15; FSEL:=RF=1 AND FXI1>-15; GSEL:=RG=1 AND GLD1>-15;
ISEL:=RI=1 AND IBIT1>-15; QSEL:=RQ=1 AND QQQ1>-15; LSEL:=RL=1 AND SLV1>-15;
OSEL:=RO=1 AND SOXX1>-15; SSEL:=RS=1 AND SPY1>-15; XSEL:=RX=1 AND XLE1>-15;
DX:=IF(DSEL,DRAML*100,0); FX:=IF(FSEL,FXIL*100,0); GX:=IF(GSEL,GLDL*100,0);
IX:=IF(ISEL,IBITL*100,0); QX:=IF(QSEL,QQQL*100,0); LX:=IF(LSEL,SLVL*100,0);
OX:=IF(OSEL,SOXXL*100,0); SX:=IF(SSEL,SPYL*100,0); XX:=IF(XSEL,XLEL*100,0);

{四、第1至5名在左列，第6至9名在右列；两列共享同一全局排序}
DY:=IF(RD<=5,0.08+(RD-1)*0.18,0.08+(RD-6)*0.18);
FY:=IF(RF<=5,0.08+(RF-1)*0.18,0.08+(RF-6)*0.18);
GY:=IF(RG<=5,0.08+(RG-1)*0.18,0.08+(RG-6)*0.18);

OY:=IF(RO<=5,0.08+(RO-1)*0.18,0.08+(RO-6)*0.18);
SY:=IF(RS<=5,0.08+(RS-1)*0.18,0.08+(RS-6)*0.18);
XY:=IF(RX<=5,0.08+(RX-1)*0.18,0.08+(RX-6)*0.18);
